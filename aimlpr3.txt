3 coordinates = {

    'B': (1, 2),

4 def heuristic(node, goal):
    """
    Euclidean distance between node and goal.
    """
    x1, y1 = coordinates[node]
    x2, y2 = coordinates[goal]

    return math.sqrt((x1 - x2) ** 2 + (y1 - y2) ** 2)

5 def reconstruct_path(parent, start, goal):
    """
    Reconstruct the path from start to goal.
    """
    path = []
    current = goal

    while current is not None:
        path.append(current)

        if current == start:
            break

        current = parent[current]

    path.reverse()

    if path[0] != start:
        return None

    return path

6 def path_cost(path, graph):
    if path is None:
        return float('inf')

    cost = 0

    for i in range(len(path) - 1):
        cost += graph[path[i]][path[i + 1]]

    return cost

7 def greedy_best_first_search(graph, start, goal):
    """
    Greedy Best-First Search

    Priority:
        f(n) = h(n)

    Returns:
        path, cost, nodes_expanded
    """

    priority_queue = []

    # (heuristic, node)
    heapq.heappush(
        priority_queue,
        (heuristic(start, goal), start)
    )

    parent = {start: None}
    visited = set()

    nodes_expanded = 0

    while priority_queue:

        _, current = heapq.heappop(priority_queue)

        if current in visited:
            continue

        visited.add(current)
        nodes_expanded += 1

        # Goal reached
        if current == goal:
            path = reconstruct_path(parent, start, goal)
            return path, path_cost(path, graph), nodes_expanded

        # Explore neighbors
        for neighbor in graph[current]:

            if neighbor not in visited:

                if neighbor not in parent:
                    parent[neighbor] = current

                heapq.heappush(
                    priority_queue,
                    (heuristic(neighbor, goal), neighbor)
                )

    return None, float('inf'), nodes_expanded

8 def a_star_search(graph, start, goal):
    """
    A* Search

    Priority:
        f(n) = g(n) + h(n)

    Returns:
        path, cost, nodes_expanded
    """

    # (f_score, g_score, node)
    priority_queue = []

    g_score = {start: 0}
    parent = {start: None}

    heapq.heappush(
        priority_queue,
        (heuristic(start, goal), 0, start)
    )

    nodes_expanded = 0

    while priority_queue:

        f_current, current_g, current = heapq.heappop(priority_queue)

        # Ignore outdated queue entries
        if current_g != g_score.get(current, float('inf')):
            continue

        nodes_expanded += 1

        # Goal reached
        if current == goal:
            path = reconstruct_path(parent, start, goal)
            return path, g_score[goal], nodes_expanded

        # Explore neighbors
        for neighbor, edge_cost in graph[current].items():

            tentative_g = g_score[current] + edge_cost

            if tentative_g < g_score.get(
                neighbor,
                float('inf')
            ):

                parent[neighbor] = current
                g_score[neighbor] = tentative_g

                f_score = (
                    tentative_g +
                    heuristic(neighbor, goal)
                )

                heapq.heappush(
                    priority_queue,
                    (f_score, tentative_g, neighbor)
                )

    return None, float('inf'), nodes_expanded
