# AI6: LINEAR REGRESSION AND POLYNOMIAL REGRESSION
# ============================================================

# 1. Import Libraries

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

from sklearn.datasets import fetch_california_housing
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression
from sklearn.preprocessing import PolynomialFeatures, StandardScaler
from sklearn.pipeline import make_pipeline
from sklearn.metrics import mean_squared_error, mean_absolute_error, r2_score


# 2. Load California Housing Dataset

data = fetch_california_housing(as_frame=True)

X = data.data
Y = data.target


# 3. Split Dataset into Training and Testing Sets

X_train, X_test, Y_train, Y_test = train_test_split(
    X, Y, test_size=0.2, random_state=42
)


# 4. Create and Train Linear Regression Model

lin_model = LinearRegression()

lin_model.fit(X_train, Y_train)

Y_pred_lin = lin_model.predict(X_test)


# 5. Create and Train Polynomial Regression Model

poly_model = make_pipeline(
    StandardScaler(),
    PolynomialFeatures(degree=2, include_bias=False),
    LinearRegression()
)

poly_model.fit(X_train, Y_train)

Y_pred_poly = poly_model.predict(X_test)


# 6. Define Function to Calculate Model Performance

def calculate_scores(y_actual, y_predicted):

    mse = mean_squared_error(y_actual, y_predicted)

    mae = mean_absolute_error(y_actual, y_predicted)

    r2 = r2_score(y_actual, y_predicted)

    return mse, mae, r2


# 7. Evaluate Both Models

linear_results = calculate_scores(Y_test, Y_pred_lin)

polynomial_results = calculate_scores(Y_test, Y_pred_poly)


# 8. Create Comparison Table

results = pd.DataFrame({
    "Metric": ["MSE", "MAE", "R2 Score"],
    "Linear Regression": linear_results,
    "Polynomial Regression": polynomial_results
})


# 9. Create Graphs

fig, axes = plt.subplots(1, 2, figsize=(12, 5))

# Linear Regression Graph

axes[0].scatter(Y_test, Y_pred_lin, alpha=0.3)

axes[0].plot(
    [Y_test.min(), Y_test.max()],
    [Y_test.min(), Y_test.max()]
)

axes[0].set_title(
    f"Linear Regression (R2 = {linear_results[2]:.2f})"
)

axes[0].set_xlabel("Actual Value")

axes[0].set_ylabel("Predicted Value")


# Polynomial Regression Graph

axes[1].scatter(Y_test, Y_pred_poly, alpha=0.3)

axes[1].plot(
    [Y_test.min(), Y_test.max()],
    [Y_test.min(), Y_test.max()]
)

axes[1].set_title(
    f"Polynomial Regression (R2 = {polynomial_results[2]:.2f})"
)

axes[1].set_xlabel("Actual Value")

axes[1].set_ylabel("Predicted Value")


plt.tight_layout()


# ============================================================
# 10. DISPLAY ALL OUTPUTS AT THE END
# ============================================================

print("\n========== CALIFORNIA HOUSING DATASET ==========")

print("\nFirst 5 Rows:")
print(X.head())

print("\nFirst 5 Target Values:")
print(Y.head())

print("\nDataset Shape:")
print(X.shape)


print("\n========== TRAINING AND TESTING DATA ==========")

print("Training Set Shape:", X_train.shape)

print("Testing Set Shape:", X_test.shape)


print("\n========== MODEL COMPARISON ==========")

print(results.to_string(index=False, float_format="%.4f"))


print("\n========== FINAL RESULT ==========")

print("Linear Regression R2 Score:", round(linear_results[2], 4))

print("Polynomial Regression R2 Score:", round(polynomial_results[2], 4))

print("\nExecution Completed Successfully!")


# 11. Display Graphs

plt.show()
